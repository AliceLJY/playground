node tools/build.mjs
# 另一个终端在仓根启动：python3 -m http.server 8763 --bind 127.0.0.1
NODE_PATH="$(npm root -g)" node tools/browser-check.cjs <仓外输出目录> http://127.0.0.1:8763/gz-roach/dist/index.html

- 构建产物：`dist/index.html`，642093 字节（小于 1 MiB），SHA-256 `ce93e35ac0b1f577ac2c71032656e6fc3d3841925ee0d35750f755273beb1e88`。dist 不进仓。
